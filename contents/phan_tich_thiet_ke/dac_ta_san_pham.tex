\section{Đặc tả sản phẩm}

Sản phẩm IoT Gateway dạng mô-đun được thiết kế nhằm tối ưu hóa tính linh hoạt và khả năng cấu hình cho các kịch bản ứng dụng IoT đa dạng:

\subsection{Thông số nhân xử lý}
Hệ thống áp dụng kiến trúc Dual-MCU Master-Slave để phân tách nhiệm vụ giữa miền mạng diện rộng (WAN) và miền thu thập dữ liệu nội bộ (LAN).
\begin{itemize}
    \item \textbf{Vi điều khiển chính (Master WAN-MCU):} ESP32-S3-WROOM-1U-N16R8.
    \begin{itemize}
        \item CPU: Xtensa® lõi kép 32-bit LX7, xung nhịp lên đến 240 MHz.
        \item Bộ nhớ: 16 MB Flash và 8 MB PSRAM tích hợp.
        \item Kết nối không dây: Tích hợp Wi-Fi 2.4 GHz và Bluetooth Low Energy (BLE).
        \item Vai trò: Điều phối hệ thống, quản lý kết nối Cloud và thực hiện cập nhật FOTA cho cả hai MCU.
    \end{itemize}
    \item \textbf{Vi điều khiển phụ (Slave LAN-MCU):} ESP32-S3-WROOM-1U-N16R8.
    \begin{itemize}
        \item Vai trò: Quản lý các giao thức mạng nội bộ, thu thập dữ liệu cảm biến và lưu trữ cục bộ.
    \end{itemize}
    \item \textbf{Giao tiếp nội bộ (Inter-MCU):} 
    \begin{itemize}
        \item Quad SPI: Kênh dữ liệu tốc độ cao để trao đổi khung tin dữ liệu và yêu cầu.
        \item UART: Kênh chuyên biệt dành cho luồng nạp Firmware liên chip (Passthrough Flashing).
        \item GPIO: Các đường tín hiệu điều khiển và ngắt (Boot, Reset, Interrupt) để đồng bộ trạng thái.
    \end{itemize}
\end{itemize}

\subsection{Khả năng kết nối và Mở rộng mô-đun}
\begin{itemize}
    \item \textbf{Khe cắm mở rộng:} Gồm 03 khe cắm chuẩn hóa (01 WAN Module, 02 LAN Module) hỗ trợ thay thế linh hoạt.
    \item \textbf{Giao diện tại khe cắm:} Hỗ trợ đầy đủ các giao tiếp UART, SPI, I2C, USB 2.0 và các chân GPIO chức năng.
    \item \textbf{Các mô-đun Adapter hỗ trợ:}
    \begin{itemize}
        \item WAN: SIM7600G-H (LTE 4G/NB-IoT), Ethernet\dots
        \item LAN: Zigbee, LoRa, RS-485, CAN Bus, và mô-đun đa giao thức Bluetooth/Thread\dots
    \end{itemize}
    \item \textbf{Lưu trữ cục bộ:} Khe cắm thẻ nhớ Micro SD hỗ trợ giao tiếp SDIO 4-bit phục vụ việc ghi nhật ký dữ liệu ngoại tuyến.
\end{itemize}

\subsection{Đặc tả Hệ thống nguồn và Bảo vệ}
Mạng phân phối nguồn (PDN) được thiết kế phân tầng để đảm bảo tính toàn vẹn năng lượng cho toàn hệ thống.
\begin{itemize}
    \item \textbf{Nguồn đầu vào:} DC 12V, phù hợp với hạ tầng công nghiệp.
    \item \textcolor{red}{\textbf{Các mức điện áp hệ thống:} Cung cấp đồng thời các ray nguồn +5V0\_SYS, +5V0, +3V3, và +1V8 phục vụ các khối chức năng và mô-đun cắm thêm.}
    \item \textbf{Cơ chế Bảo vệ phần cứng:}
    \begin{itemize}
        \item Electronic Fuse (eFuse): Sử dụng eFuse để bảo vệ quá dòng và quá áp, giới hạn dòng đỉnh tại có điều khiển.
        \item Chống tĩnh điện (ESD): Trang bị TVS Diode tại đầu vào và TVS chuyên dụng cho các đường dữ liệu.
        \item Lọc nhiễu: Hệ thống tụ Decoupling đa tầng đặt sát chân linh kiện để triệt tiêu nhiễu cao tần.
    \end{itemize}
\end{itemize}

\subsection{Giao diện người dùng và Đặc điểm cơ khí}
\begin{itemize}
    \item \textcolor{red}{\textbf{Hiển thị:} Màn hình OLED 0.96 inch (Driver SH1107), giao tiếp I2C để thông báo trạng thái.}
    \item \textbf{Chỉ thị trạng thái:} Hệ thống đèn LED báo hiệu trạng thái nguồn, kết nối LAN/WAN và trạng thái máy chủ.
    \item \textbf{Quản lý thời gian:} IC RTC kèm pin dự phòng CR2032 giúp duy trì thời gian thực khi mất điện.
    \item \textbf{Tương tác vật lý:} Nút nhấn Reset, Boot và nút nhấn điều khiển nguồn tích hợp.
\end{itemize}

\subsection{Đặc tả Phần mềm}
\begin{itemize}
    \item \textbf{Hệ điều hành nhúng:} Chạy trên nền tảng FreeRTOS, hỗ trợ lập lịch đa nhiệm và xử lý song song.
    \item \textbf{Giao thức truyền dữ liệu:} Hỗ trợ MQTT, HTTP, và CoAP theo cấu hình linh hoạt.
    \item \textbf{Khả năng cấu hình:} Sử dụng công cụ Configuration Tools trên PC để thiết lập tham số (chu kỳ gửi, thông số mạng, lựa chọn mô-đun) qua USB/UART mà không cần nạp lại Firmware.
    \item \textbf{Tính an toàn và ổn định:} Tích hợp Watchdog Timer tự động phục hồi khi có sự cố treo hệ thống và cơ chế lưu đệm hàng đợi (Queue) khi mất kết nối mạng.
\end{itemize}
